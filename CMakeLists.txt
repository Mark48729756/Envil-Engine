if(ANVIL_PHYSICS_ONLY)
  include(CTest)
  include(FetchContent)
  if(NOT TARGET glm::glm)
    FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glm)
  endif()
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
  endif()
  add_library(anvil_physics STATIC engine/src/PhysicsWorld.cpp engine/src/physics/PhysicsBackend.cpp)
  target_include_directories(anvil_physics PUBLIC engine/include PRIVATE engine/src)
  target_link_libraries(anvil_physics PUBLIC glm::glm)
  if(ANVIL_WITH_JOLT)
    include(cmake/AnvilJolt.cmake)
    target_sources(anvil_physics PRIVATE engine/src/JoltPhysicsWorld.cpp engine/src/physics/JoltBackend.cpp)
    target_link_libraries(anvil_physics PUBLIC Jolt)
    target_compile_definitions(anvil_physics PUBLIC ANVIL_WITH_JOLT=1)
  endif()
  add_executable(anvil_physics_backend_tests tests/PhysicsBackendTests.cpp)
  target_link_libraries(anvil_physics_backend_tests PRIVATE anvil_physics)
  add_test(NAME anvil_physics_backend_tests COMMAND anvil_physics_backend_tests)
  return()
endif()

FetchContent_Declare(glad GIT_REPOSITORY https://github.com/Dav1dde/glad.git GIT_TAG v2.0.6 GIT_SHALLOW TRUE SOURCE_SUBDIR cmake)
# editor-only dependencies: Dear ImGui (docking branch) and nanosvg (SVG -> texture for the editor icons).
# SOURCE_SUBDIR points at a non-existing folder so FetchContent only downloads them; we build the few files we need ourselves.
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.93.0-docking GIT_SHALLOW TRUE SOURCE_SUBDIR _none)
FetchContent_Declare(nanosvg GIT_REPOSITORY https://github.com/memononen/nanosvg.git GIT_TAG master GIT_SHALLOW TRUE SOURCE_SUBDIR _none)
FetchContent_MakeAvailable(glfw glm glad imgui nanosvg)
